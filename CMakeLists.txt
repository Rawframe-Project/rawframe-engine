cmake_minimum_required(VERSION 3.28)

project(rawframe VERSION 0.1.0 LANGUAGES C CXX)

# Strict C++23: no GNU extensions, and no C++20 module scanning (Rawframe uses
# headers, and scanning needs a clang-scan-deps that not every toolchain ships).
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

include(cmake/rawframe.cmake)

option(RAWFRAME_BUILD_TESTS "Build the tests" ON)

enable_testing()

if(RAWFRAME_BUILD_TESTS)
    add_subdirectory(tests/harness)
endif()

# The web build (cmake/wasm32-wasi.cmake) is the modules tools/web_modules.txt
# lists, the third-party code they need, and the web client host, which is
# listed there too.
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
    file(STRINGS "${PROJECT_SOURCE_DIR}/tools/web_modules.txt" rawframe_web_lines REGEX "^[a-z]")
    string(JOIN " " rawframe_web_modules ${rawframe_web_lines})
    separate_arguments(rawframe_web_modules)
    # Third-party code a web module needs, built the same way.
    if("kest" IN_LIST rawframe_web_modules)
        add_subdirectory(third_party/kest)
    endif()
    if("physics2d" IN_LIST rawframe_web_modules)
        add_subdirectory(third_party/maul2d)
    endif()
    if("physics3d" IN_LIST rawframe_web_modules)
        add_subdirectory(third_party/maul3d)
    endif()
    if("content" IN_LIST rawframe_web_modules)
        add_subdirectory(third_party/zstd)
    endif()
    if("signature" IN_LIST rawframe_web_modules)
        include(third_party/openssl_wasm.cmake)
    endif()
    foreach(module IN LISTS rawframe_web_modules)
        if(module STREQUAL "web_client")
            add_subdirectory(hosts/web_client)
        else()
            add_subdirectory(modules/${module})
        endif()
    endforeach()
    return()
endif()
